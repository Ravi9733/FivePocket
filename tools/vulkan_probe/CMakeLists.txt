if(RECOMP_GLSLC OR RECOMP_SPIRV_VAL)
    if(NOT EXISTS "${RECOMP_GLSLC}" OR IS_DIRECTORY "${RECOMP_GLSLC}" OR
       NOT EXISTS "${RECOMP_SPIRV_VAL}" OR IS_DIRECTORY "${RECOMP_SPIRV_VAL}")
        message(FATAL_ERROR "Both explicit shader compiler and validator files are required")
    endif()
    set(constant_modules)
    foreach(stage vert frag)
        set(source "${CMAKE_CURRENT_SOURCE_DIR}/constant_point.${stage}")
        set(module "${CMAKE_CURRENT_BINARY_DIR}/generic-constant-point.${stage}.spv")
        # Publish only after validation. Failed validation must not leave a
        # fresh final output that a later incremental build could accept.
        add_custom_command(OUTPUT "${module}"
            COMMAND "${RECOMP_GLSLC}" --target-env=vulkan1.1 "${source}" -o "${module}.unvalidated"
            COMMAND "${RECOMP_SPIRV_VAL}" --target-env vulkan1.1 "${module}.unvalidated"
            COMMAND "${CMAKE_COMMAND}" -E rename "${module}.unvalidated" "${module}"
            DEPENDS "${source}" "${RECOMP_GLSLC}" "${RECOMP_SPIRV_VAL}"
            VERBATIM)
        list(APPEND constant_modules "${module}")
    endforeach()
    add_custom_target(generic_constant_shaders DEPENDS ${constant_modules})
endif()
